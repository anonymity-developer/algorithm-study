import java.util.*;

class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();

        for (int tc=1; tc<=T; tc++) {
            int N = sc.nextInt();
            int M = sc.nextInt();

            boolean[][] reach = new boolean[N+1][N+1];

            for (int i=0; i<M; i++) {
                int a = sc.nextInt();
                int b = sc.nextInt();

                reach[a][b] = true;
            }

            // Floyd-Warshall
            for (int k=1; k<=N; k++) {
                for (int i=1; i<=N; i++) {
                    if (!reach[i][k]) continue;

                    for (int j=1; j<=N; j++) {
                        if (reach[k][j])
                            reach[i][j] = true;
                    }
                }
            }

            int ans = 0;

            for (int i=1; i<=N; i++) {
                int cnt = 0;

                for (int j=1; j<=N; j++) {
                    if (i == j) continue;

                    if (reach[i][j] || reach[j][i])
                        cnt++;
                }

                if (cnt == N-1)
                    ans++;
            }

            System.out.println("#" + tc + " " + ans);
        }
    }
}
